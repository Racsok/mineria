import pandas as pd

df = pd.read_csv("candidato_1.csv")          # ojo con sep y encoding

# ── 1. Los numeros ────────────────────────────────────────────
print(df.shape)
print(df.dtypes.value_counts())

# ── 2. El perfilado (sesion 4) ────────────────────────────────
perfilar(df)                                  # tipo, distintos, faltan_%, moda, min/mediana/media/max
print("duplicados:", df.duplicated().sum())
print("faltantes totales: {:.1%}".format(df.isna().mean().mean()))
print(df.isna().mean().sort_values(ascending=False).head(5) * 100)

# ── 3. El objetivo ────────────────────────────────────────────
OBJ = "mi_columna_objetivo"
print(df[OBJ].value_counts(normalize=True) * 100)     # si es categorica
# print(df[OBJ].describe())                           # si es continua

# ── 4. Atipicos: error o hallazgo (sesion 4) ──────────────────
for col in ["columna_a", "columna_b"]:
    n_at, bajo, alto = atipicos_iqr(df[col])
    print(f"{col:22s} atipicos={n_at:5d}   rango normal=[{bajo}, {alto}]")

# ── 5. La integracion (sesion 3) ──────────────────────────────
otra = pd.read_csv("segunda_fuente.csv")
integrado = unir(df, otra, "llave", nombre="+ segunda fuente")   # cuenta filas antes y despues

Los tres candidatos

En orden de preferencia: el candidato 1 es el que de verdad quieren. Se revisa en ese orden y se para en el primero que cumple, así que el orden importa.

Los tres tienen que ser distintos de verdad —no tres recortes del mismo archivo ni el mismo indicador en tres años— y no pueden salir los tres de Kaggle: al menos dos fuentes distintas y al menos una institucional.
